1.	Для начала потребуется сконфигурировать nftables на BR-RTR.

`tee /etc/nftables.conf <<'EOF'`

`#!/usr/sbin/nft -f`


`flush ruleset`

`table inet filter {`

        chain input {

                type filter hook input priority filter;

        }

        chain forward {

                type filter hook forward priority filter;

        }

        chain output {

                 type filter hook output priority filter;

        }
`}`

`table inet nat {`

    chain PREROUTING {

    type nat hook prerouting priority filter;

    ip daddr 172.16.5.2 tcp dport 8080 dnat ip to 192.168.0.2:8080

    ip daddr 172.16.5.2 tcp dport 2026 dnat ip to 192.168.0.2:2026

    }


    chain POSTROUTING {

    type nat hook postrouting priority srcnat;

    oifname "ens192" masquerade

    }

`}`

`EOF`
Важно! Нужно обратить внимание на правило с портом 2026, это правило для SSH, в данном пункте присутствуют сразу два правила в одном файле конфигурации.

`systemctl restart nftables` - перезапуск службы nftables.

4.	Теперь нужно настроить nftables на маршрутизаторе HQ-RTR.

`tee /etc/nftables.conf <<'EOF'`

`#!/usr/sbin/nft -f`

`flush ruleset`

`table inet filter {`
        chain input {
                type filter hook input priority filter;
        }
        chain forward {
                type filter hook forward priority filter;
        }
        chain output {
                 type filter hook output priority filter;
        }
`}`

`table inet nat {`
    chain PREROUTING {
    type nat hook prerouting priority filter;
    ip daddr 172.16.4.2 tcp dport 8080 dnat ip to 192.168.100.2:80
    ip daddr 172.16.4.2 tcp dport 2026 dnat ip to 192.168.100.2:2026
    }

    chain POSTROUTING {
    type nat hook postrouting priority srcnat;
    oifname "ens192" masquerade
    }
`}`
`EOF`

Важно! Нужно обратить внимание на правило с портом 2026, это правило для SSH, в данном пункте присутствуют сразу два правила в одном файле конфигурации.

`systemctl restart nftables` - перезапуск службы nftables.
