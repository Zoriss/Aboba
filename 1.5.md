## Настройку делаем на HQ-SRV

Скачиваем пакет bind9

    apt install bind9 -y

## Основная конфигурация

    nano /etc/bind/named.conf.options

<img width="799" height="427" alt="image" src="https://github.com/user-attachments/assets/dfb623cb-ec73-477d-a1bc-579895ca63c2" />

    forwarders { 8.8.8.8; };
    listen-on port 53 { any; };

.

    dnssec-validation no;

    allow-query { any; };

    listen-on-v6 { none; };
    };

## Обозначение зон

    nano /etc/bind/named.conf.default-zones

Удаляем или комментируем имеющиеся зоны и добавляем свои.

<img width="598" height="441" alt="image" src="https://github.com/user-attachments/assets/d92105b8-3026-41b4-bd6f-13d6962721a5" />



    zone "au-team.irpo" {
    type master;
    file "/etc/bind/au-team.db";
    };

    zone "100.168.192.in-addr.arpa" {
    type master;
    file "/etc/bind/au-team_rev.db";
    };

    zone "200.168.192.in-addr.arpa" {
    type master;
    file "/etc/bind/au-team_rev2.db";
    };

    zone "0.168.192.in-addr.arpa" {
    type master;
    file "/etc/bind/au-team_rev3.db";
    };

## Настройка зон

Копируем шаблоны для зон из существующего файла

    cp /etc/bind/db.empty /etc/bind/au-team.db

.

    cp /etc/bind/db.empty /etc/bind/au-team_rev.db

.

    cp /etc/bind/db.empty /etc/bind/au-team_rev2.db

.

    cp /etc/bind/db.empty /etc/bind/au-team_rev3.db

Перейдём к редактированию конфигов

    nano /etc/bind/au-team.db

<img width="587" height="409" alt="image" src="https://github.com/user-attachments/assets/54d677a7-e869-4185-a981-8130757a9e09" />

    au-team.irpo

.

    IN A 192.168.100.2
    hq-srv IN A 192.168.100.2
    hq-cli IN A 192.168.200.2
    hq-rtr IN A 192.168.100.1
    hq-rtr IN A 192.168.200.1
    br-rtr IN A 192.168.0.1
    br-srv IN A 192.168.0.2
    web IN A 172.16.10.1
    docker IN A 172.16.20.1

.

    nano /etc/bind/au-team_rev.db

## Настройка на ВСЕХ машинах, кроме ISP

Стираем всё и прописываем ip-адрес сервера, на котором настроили DNS, в нашем случае HQ-SRV

    nano /etc/resolv.conf
.
